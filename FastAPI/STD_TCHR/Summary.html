3.1. app/database/db.py

This file sets up the database connection and provides a dependency for getting a database session in your API routes.

# app/database/db.py

from sqlalchemy import create_engine
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker

# SQLite database URL
SQLALCHEMY_DATABASE_URL = "sqlite:///./std_tchr.db"

# Create the SQLAlchemy engine
# connect_args={"check_same_thread": False} is needed for SQLite to allow multiple threads
# to interact with the database, which FastAPI (Uvicorn) often does.
engine = create_engine(
    SQLALCHEMY_DATABASE_URL, connect_args={"check_same_thread": False}
)

# Create a SessionLocal class
# Each instance of SessionLocal will be a database session.
# The 'autocommit=False' means we have to explicitly commit changes.
# The 'autoflush=False' means data isn't flushed to the database until commit or query.
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

# Base class for your ORM models
Base = declarative_base()

# Dependency to get a database session
# This function will be used with FastAPI's Depends()
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

# app/crud/student_crud.py

from sqlalchemy.orm import Session
from sqlalchemy.exc import IntegrityError # For handling unique constraints

from app.models.students import Student
from app.schemas.std_schema import Std_Create, Std_Update

# Get a single student by ID
def get_student(db: Session, student_id: int):
    return db.query(Student).filter(Student.std_id == student_id).first()

# Get a student by email (useful for uniqueness checks)
def get_student_by_email(db: Session, email: str):
    return db.query(Student).filter(Student.std_email == email).first()

# Get multiple students with pagination
def get_students(db: Session, skip: int = 0, limit: int = 100):
    return db.query(Student).offset(skip).limit(limit).all()

# Create a new student
def create_student(db: Session, student: Std_Create):
    db_student = Student(
        std_name=student.std_name,
        std_email=student.std_email,
        department=student.department
    )
    try:
        db.add(db_student)
        db.commit()
        db.refresh(db_student) # Refresh to get the generated std_id
        return db_student
    except IntegrityError:
        db.rollback()
        # You would typically raise an HTTPException here in the router
        # or a custom error that the router can catch.
        raise ValueError("Student with this email already exists.")


# Update an existing student
def update_student(db: Session, student_id: int, student: Std_Update):
    db_student = db.query(Student).filter(Student.std_id == student_id).first()
    if db_student:
        update_data = student.model_dump(exclude_unset=True) # Use model_dump to get only provided fields
        for key, value in update_data.items():
            setattr(db_student, key, value)
        try:
            db.add(db_student)
            db.commit()
            db.refresh(db_student)
            return db_student
        except IntegrityError:
            db.rollback()
            raise ValueError("Updated email already exists for another student.")
    return None # Student not found

# Delete a student
def delete_student(db: Session, student_id: int):
    db_student = db.query(Student).filter(Student.std_id == student_id).first()
    if db_student:
        db.delete(db_student)
        db.commit()
        return db_student # Return the deleted student object (or True/False)
    return None # Student not found

3.13. app/crud/enrollment_crud.py

Functions for managing Student-Course enrollments.

# app/crud/enrollment_crud.py

from sqlalchemy.orm import Session
from sqlalchemy.exc import IntegrityError

from app.models.students import Student
from app.models.courses import Course
from app.schemas.enrollment_schema import EnrollmentCreate # Using the Pydantic schema

# Function to enroll a student in a course
def enroll_student_in_course(db: Session, student_id: int, course_id: int):
    # Get the student and course ORM objects
    db_student = db.query(Student).filter(Student.std_id == student_id).first()
    db_course = db.query(Course).filter(Course.crs_id == course_id).first()

    if not db_student:
        raise ValueError(f"Student with ID {student_id} not found.")
    if not db_course:
        raise ValueError(f"Course with ID {course_id} not found.")

    # Check if enrollment already exists
    if db_course in db_student.courses:
        raise ValueError(f"Student {student_id} is already enrolled in course {course_id}.")

    # Add the course to the student's courses relationship
    # SQLAlchemy handles the association table insertion automatically
    try:
        db_student.courses.append(db_course)
        db.commit()
        db.refresh(db_student)
        # We return a simple dictionary or a Pydantic model representation
        # as the association table doesn't have its own ORM model usually.
        return {"student_id": student_id, "course_id": course_id, "status": "enrolled"}
    except IntegrityError:
        db.rollback()
        raise ValueError("Enrollment failed due to a database integrity error (e.g., duplicate).")


# Function to get all enrollments for a student
def get_student_enrollments(db: Session, student_id: int):
    db_student = db.query(Student).filter(Student.std_id == student_id).first()
    if not db_student:
        return None
    # Returns a list of Course objects
    return db_student.courses

# Function to get all students enrolled in a course
def get_course_enrollments(db: Session, course_id: int):
    db_course = db.query(Course).filter(Course.crs_id == course_id).first()
    if not db_course:
        return None
    # Returns a list of Student objects
    return db_course.students

# Function to unenroll a student from a course
def unenroll_student_from_course(db: Session, student_id: int, course_id: int):
    db_student = db.query(Student).filter(Student.std_id == student_id).first()
    db_course = db.query(Course).filter(Course.crs_id == course_id).first()

    if not db_student:
        raise ValueError(f"Student with ID {student_id} not found.")
    if not db_course:
        raise ValueError(f"Course with ID {course_id} not found.")

    if db_course not in db_student.courses:
        raise ValueError(f"Student {student_id} is not enrolled in course {course_id}.")

    try:
        # Remove the course from the student's courses relationship
        db_student.courses.remove(db_course)
        db.commit()
        db.refresh(db_student)
        return {"student_id": student_id, "course_id": course_id, "status": "unenrolled"}
    except Exception as e:
        db.rollback()
        raise ValueError(f"Unenrollment failed: {e}")

3.14. app/routers/std_router.py

FastAPI routes for Student API.

# app/routers/std_router.py

from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.orm import Session
from typing import List

from app.database.db import get_db
from app.schemas.std_schema import Student, Std_Create, Std_Update
from app.crud import student_crud

router = APIRouter(
    prefix="/students",
    tags=["students"],
    responses={404: {"description": "Student not found"}},
)

@router.post("/", response_model=Student, status_code=status.HTTP_201_CREATED)
def create_student_route(student: Std_Create, db: Session = Depends(get_db)):
    db_student_email = student_crud.get_student_by_email(db, email=student.std_email)
    if db_student_email:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Student with this email already registered"
        )
    try:
        return student_crud.create_student(db=db, student=student)
    except ValueError as e:
        raise HTTPException(status_code=status.HTTP_400_BAD_REQUEST, detail=str(e))


@router.get("/", response_model=List[Student])
def read_students_route(skip: int = 0, limit: int = 100, db: Session = Depends(get_db)):
    students = student_crud.get_students(db, skip=skip, limit=limit)
    return students

@router.get("/{student_id}", response_model=Student)
def read_student_route(student_id: int, db: Session = Depends(get_db)):
    db_student = student_crud.get_student(db, student_id=student_id)
    if db_student is None:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Student not found")
    return db_student

@router.put("/{student_id}", response_model=Student)
def update_student_route(student_id: int, student: Std_Update, db: Session = Depends(get_db)):
    # Check if the new email (if provided) conflicts with another student
    if student.std_email:
        existing_student_with_email = student_crud.get_student_by_email(db, email=student.std_email)
        if existing_student_with_email and existing_student_with_email.std_id != student_id:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="Email already belongs to another student"
            )
    try:
        db_student = student_crud.update_student(db, student_id=student_id, student=student)
        if db_student is None:
            raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Student not found")
        return db_student
    except ValueError as e:
        raise HTTPException(status_code=status.HTTP_400_BAD_REQUEST, detail=str(e))

@router.delete("/{student_id}", status_code=status.HTTP_204_NO_CONTENT)
def delete_student_route(student_id: int, db: Session = Depends(get_db)):
    db_student = student_crud.delete_student(db, student_id=student_id)
    if db_student is None:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Student not found")
    return None # 204 No Content typically returns no body

        if existing_teacher_with_email and existing_teacher_with_email.tchr_id != teacher_id:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="Email already belongs to another teacher"
            )
    try:
        db_teacher = tchr_crud.update_teacher(db, teacher_id=teacher_id, teacher=teacher)
        if db_teacher is None:
            raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Teacher not found")
        return db_teacher
    except ValueError as e:
        raise HTTPException(status_code=status.HTTP_400_BAD_REQUEST, detail=str(e))

@router.delete("/{teacher_id}", status_code=status.HTTP_204_NO_CONTENT)
def delete_teacher_route(teacher_id: int, db: Session = Depends(get_db)):
    db_teacher = tchr_crud.delete_teacher(db, teacher_id=teacher_id)
    if db_teacher is None:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Teacher not found")

    prefix="/enrollments",
    tags=["enrollments"],
    responses={404: {"description": "Enrollment not found"}},
)

@router.post("/", response_model=Enrollment, status_code=status.HTTP_201_CREATED)
def enroll_student(enrollment: EnrollmentCreate, db: Session = Depends(get_db)):
    try:
        # The CRUD function handles finding student/course and creating the link
        enrollment_result = enrollment_crud.enroll_student_in_course(
            db=db,
            student_id=enrollment.student_id,
            course_id=enrollment.course_id
        )
        return enrollment_result # This will be {"student_id": ..., "course_id": ..., "status": "enrolled"}
    except ValueError as e:
        raise HTTPException(status_code=status.HTTP_400_BAD_REQUEST, detail=str(e))
    except Exception as e:
        raise HTTPException(status_code=status.HTTP_500_INTERNAL_SERVER_ERROR, detail=f"An unexpected error occurred: {e}")

@router.delete("/", status_code=status.HTTP_204_NO_CONTENT)
def unenroll_student(student_id: int, course_id: int, db: Session = Depends(get_db)):
    try:
        unenroll_result = enrollment_crud.unenroll_student_from_course(
            db=db, student_id=student_id, course_id=course_id
        )
        if unenroll_result is None: # This check might be redundant if CRUD raises ValueError
             raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Enrollment not found or already deleted")
        return None # 204 No Content
    except ValueError as e:
        raise HTTPException(status_code=status.HTTP_400_BAD_REQUEST, detail=str(e))
    except Exception as e:
        raise HTTPException(status_code=status.HTTP_500_INTERNAL_SERVER_ERROR, detail=f"An unexpected error occurred: {e}")
